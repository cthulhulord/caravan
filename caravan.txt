local SNAP_COUNT = 10
local CARAVAN_CARD_Y = 2.0

function createCaravan()

    return {

        won = false
    }

------------------------------------------------------------

------------------------------------------------------------

function initializeGame()

    GAME.caravans = {}

    GAME.processing = {}

    GAME.registeredCards = {}

    GAME.gameOver = false

    GAME.attachedKings = {}

    -- Initialize discard piles
    DISCARD_PILES = {
        [1] = {},
        [2] = {}
    }

    for caravan = 1, 3 do

        GAME.caravans[caravan] = {}

        for player = 1, 2 do

            GAME.caravans[caravan][player] =
                createCaravan()

        end

    end

end

-- GET A COUNTER
------------------------------------------------------------

function getCounter(caravanNumber, player)

    local guid = COUNTERS[caravanNumber][player]

    local counter = getObjectFromGUID(guid)

    if not counter then

        print(
            "ERROR: Could not find counter " ..
            caravanNumber ..
            "/" ..
            player
        )

    end

    return counter

end


------------------------------------------------------------
-- UPDATE PHYSICAL COUNTER
------------------------------------------------------------

function updateCounter(caravanNumber, player)

    local caravan =
        GAME.caravans[caravanNumber][player]

    local counter =
        getCounter(caravanNumber, player)

    if not counter then
        return
    end

    counter.Counter.setValue(
        caravan.value
    )

end


------------------------------------------------------------
-- CARD PARSING
------------------------------------------------------------

function parseCard(card)

    if not card then
        return nil, "No card supplied."
    end


    local name = card.getName()

    if not name or name == "" then

        return nil,
            "Card has no name. " ..
            "Use names such as 8H or 10S."

    end


    name = string.upper(
        string.gsub(name, "%s+", "")
    )


    --------------------------------------------------------
    -- JOKER
    --------------------------------------------------------

    if name == "JOKER" then

        return {
            rank = 0,
            suit = nil,
            type = "joker",
            object = card
        }

    end


    --------------------------------------------------------
    -- SUIT
    --------------------------------------------------------

    local suitLetter =
        string.sub(name, -1)

    local suitTable = {

        S = "Spades",

        H = "Hearts",

        D = "Diamonds",

        C = "Clubs"
    }


    local suit =
        suitTable[suitLetter]


    if not suit then

        return nil,
            "Invalid suit in card name: " ..
            name

    end


    --------------------------------------------------------
    -- RANK
    --------------------------------------------------------

    local rankText =
        string.sub(
            name,
            1,
            #name - 1
        )


    local rank


    if rankText == "A" then

    rank = 1

elseif rankText == "J" then

    rank = 11

elseif rankText == "Q" then

    rank = 12

elseif rankText == "K" then

    rank = 13

else

    rank = tonumber(rankText)

end


    if not rank then

    return nil,
        "Invalid rank in card name: " ..
        name

end


    --------------------------------------------------------
    -- RETURN CARD DATA
    --------------------------------------------------------

    local cardType

if rank >= 1 and rank <= 10 then

    cardType = "number"

elseif rank == 11 then

    cardType = "jack"

elseif rank == 12 then

    cardType = "queen"

elseif rank == 13 then

    cardType = "king"

end


return {

    rank = rank,

    suit = suit,

    type = cardType,

    object = card,

    name = name,

    guid = nil  -- Will be set when card is registered
}

end

------------------------------------------------------------
-- CALCULATE CARAVAN VALUE
------------------------------------------------------------

function calculateCaravanValue(caravan)

    local total = 0

    for _, card in ipairs(caravan.cards) do

        if card.type == "number" then

            -- Apply King multipliers attached to this card
            local multiplier = getKingMultiplier(card)
            total = total + (card.rank * multiplier)

        end

    end

    return total

end


------------------------------------------------------------
-- GET KING MULTIPLIER FOR A CARD
------------------------------------------------------------

function getKingMultiplier(card)

    if not card or not card.guid then
        return 1
    end

    local kings = GAME.attachedKings[card.guid]
    if not kings or #kings == 0 then
        return 1
    end

    -- Each King doubles the value: 1 King = x2, 2 Kings = x4, etc.
    return math.pow(2, #kings)

end



------------------------------------------------------------
-- FIND NUMBER CARD AT POSITION (for King attachment)
------------------------------------------------------------

function findNumberCardAtPosition(caravanNumber, player, position)
    local caravan = GAME.caravans[caravanNumber][player]
    
    -- Get the counter position to calculate relative positions
    local counter = getCounter(caravanNumber, player)
    if not counter then
        return nil
    end
    
    local counterPos = counter.getPosition()
    local threshold = 1.5 -- Distance threshold to consider a card "at" a position
    
    -- Check all cards in this caravan
    for _, card in ipairs(caravan.cards) do
        if card.type == "number" and card.guid then
            local cardObj = getObjectFromGUID(card.guid)
            if cardObj then
                local cardPos = cardObj.getPosition()
                local dist = math.sqrt(
                    math.pow(cardPos.x - position.x, 2) +
                    math.pow(cardPos.y - position.y, 2) +
                    math.pow(cardPos.z - position.z, 2)
                )
                
                if dist < threshold then
                    return card
                end
            end
        end
    end
    
    return nil
end


------------------------------------------------------------
-- FIND NUMBER CARD BY SLOT
-- Returns the number card at a specific slot in a caravan
------------------------------------------------------------

function findNumberCardBySlot(caravanNumber, player, slot)
    local caravan = GAME.caravans[caravanNumber][player]
    
    for _, card in ipairs(caravan.cards) do
        if card.type == "number" and card.slot == slot then
            return card
        end
    end
    
    return nil
end


------------------------------------------------------------
-- GET SLOT FROM POSITION
-- Determines which slot a position corresponds to in a caravan
------------------------------------------------------------

function getSlotFromPosition(caravanNumber, player, position)
    local counter = getCounter(caravanNumber, player)
    if not counter then
        return nil
    end
    
    local counterPos = counter.getPosition()
    local threshold = 1.0 -- Distance threshold for slot matching
    
    -- Calculate expected positions for each slot
    local offsetX, offsetZ, spacing
    if player == 1 then
        offsetX = P1_SNAP_OFFSET_X
        offsetZ = P1_SNAP_OFFSET_Z
        spacing = P1_SNAP_SPACING
    else
        offsetX = P2_SNAP_OFFSET_X
        offsetZ = P2_SNAP_OFFSET_Z
        spacing = P2_SNAP_SPACING
    end
    
    -- Check each slot position
    for slot = 1, SNAP_COUNT do
        local expectedX = counterPos.x + offsetX
        local expectedZ = counterPos.z + offsetZ + (spacing * (slot - 1))
        
        local dist = math.sqrt(
            math.pow(position.x - expectedX, 2) +
            math.pow(position.z - expectedZ, 2)
        )
        
        -- Also check at secondary snap point height
        local distSecondary = math.sqrt(
            math.pow(position.x - expectedX, 2) +
            math.pow(position.y - SECONDARY_SNAP_Y, 2) +
            math.pow(position.z - expectedZ, 2)
        )
        
        if dist < threshold or distSecondary < threshold then
            return slot
        end
    end
    
    return nil
end


------------------------------------------------------------
-- SEND CARD TO DISCARD PILE
------------------------------------------------------------

function sendCardToDiscard(cardObject, player)
    if not cardObject then
        return
    end
    
    -- Use the dedicated discard pile object
    local discardPile = getObjectFromGUID(DISCARD_PILE_GUID)
    
    if discardPile then
        local discardPos = discardPile.getPosition()
        discardPos.y = discardPos.y + 0.5  -- Slightly above the pile
        cardObject.setPositionSmooth(discardPos)
        
        -- Track in discard pile
        local guid = cardObject.getGUID()
        if not DISCARD_PILES[player] then
            DISCARD_PILES[player] = {}
        end
        table.insert(DISCARD_PILES[player], guid)
        
        print("[CARAVAN] Card sent to discard pile: " .. cardObject.getName())
    else
        print("[CARAVAN] WARNING: Could not find discard pile object with GUID: " .. DISCARD_PILE_GUID)
        
        -- Fallback to offset from caravan zone
        local zoneGUID = ZONES[1][player]
        local zone = getObjectFromGUID(zoneGUID)
        if zone then
            local zonePos = zone.getPosition()
            local discardPos = {
                x = zonePos.x + DISCARD_OFFSET_X,
                y = DISCARD_CARD_Y,
                z = zonePos.z + DISCARD_OFFSET_Z
            }
            cardObject.setPositionSmooth(discardPos)
        end
    end
end





------------------------------------------------------------
-- ATTACH KING TO NUMBER CARD
------------------------------------------------------------

function attachKingToCard(baseCard, kingCard)
    if not baseCard or not baseCard.guid then
        return false, "Base card must have a guid"
    end
    
    if not kingCard or not kingCard.guid then
        return false, "King card must have a guid"
    end
    
    -- Initialize the attached kings table for this base card if needed
    if not GAME.attachedKings[baseCard.guid] then
        GAME.attachedKings[baseCard.guid] = {}
    end
    
    -- Add the king to the attachment list
    table.insert(GAME.attachedKings[baseCard.guid], kingCard.guid)
    
    return true
end


------------------------------------------------------------
-- REMOVE KING ATTACHMENTS FROM CARD
------------------------------------------------------------

function removeKingAttachments(baseCardGuid)
    if not baseCardGuid then
        return
    end
    
    if GAME.attachedKings[baseCardGuid] then
        GAME.attachedKings[baseCardGuid] = {}
    end
end


------------------------------------------------------------
-- REMOVE CARD AND ATTACHED KINGS FROM CARAVAN
------------------------------------------------------------

function removeCardAndAttachments(caravanNumber, player, cardToRemove)
    if not cardToRemove or not cardToRemove.guid then
        return false
    end
    
    local caravan = GAME.caravans[caravanNumber][player]
    
    -- Remove the card from the caravan cards table
    for i = #caravan.cards, 1, -1 do
        if caravan.cards[i].guid == cardToRemove.guid then
            table.remove(caravan.cards, i)
            break
        end
    end
    
    -- Remove any Kings attached to this card
    local attachedKings = GAME.attachedKings[cardToRemove.guid] or {}
    
    -- Remove each attached King from the caravan cards table
    for _, kingGuid in ipairs(attachedKings) do
        for i = #caravan.cards, 1, -1 do
            if caravan.cards[i].guid == kingGuid then
                table.remove(caravan.cards, i)
                break
            end
        end
    end
    
    -- Clean up the attachment tracking
    removeKingAttachments(cardToRemove.guid)
    
    -- Remove from registered cards
    GAME.registeredCards[cardToRemove.guid] = nil
    
    -- Remove any Kings that were attached to this card from registeredCards
    if GAME.attachedKings[cardToRemove.guid] then
        for _, kingGuid in ipairs(GAME.attachedKings[cardToRemove.guid]) do
            GAME.registeredCards[kingGuid] = nil
        end
    end
    
    return true
end

------------------------------------------------------------
-- CHECK FIRST CARD
------------------------------------------------------------

function validateFirstCard(
    caravan,
    card
)

    if #caravan.cards > 0 then

        return false,
            "Caravan already contains cards."

    end


    if card.type ~= "number" then

        return false,
            "The first card must be an Ace or " ..
            "numbered card."

    end


    return true

end


------------------------------------------------------------
-- VALIDATE SECOND CARD
------------------------------------------------------------

function validateSecondCard(
    caravan,
    card
)

    local first =
        caravan.cards[1]


    if card.rank == first.rank then

        return false,
            "Two cards of the same value " ..
            "cannot be consecutive."

    end


    if card.rank > first.rank then

        caravan.direction = "ascending"

    else

        caravan.direction = "descending"

    end


    return true

end


------------------------------------------------------------
-- VALIDATE SUBSEQUENT CARD
------------------------------------------------------------

function validateSubsequentCard(
    caravan,
    card
)

    --------------------------------------------------------
    -- FIND THE MOST RECENT NUMERICAL CARD
    --
    -- Queens do not participate in numerical ordering.
    --------------------------------------------------------

    local previous = nil

    for i = #caravan.cards, 1, -1 do

        if caravan.cards[i].type == "number" then

            previous =
                caravan.cards[i]

            break

        end

    end


    --------------------------------------------------------
    -- A numerical card must have a previous
    -- numerical card to compare against.
    --------------------------------------------------------

    if not previous then

        return false,
            "Could not find a previous numerical card."

    end


    --------------------------------------------------------
    -- SAME VALUE IS NEVER ALLOWED
    --------------------------------------------------------

    if card.rank == previous.rank then

        return false,
            "Two cards of the same value " ..
            "cannot be consecutive."

    end


    --------------------------------------------------------
    -- CHECK NUMERICAL DIRECTION
    --------------------------------------------------------

    local directionOK = false


    if caravan.direction == "ascending" then

        if card.rank > previous.rank then

            directionOK = true

        end


    elseif caravan.direction == "descending" then

        if card.rank < previous.rank then

            directionOK = true

        end

    end


    --------------------------------------------------------
    -- CHECK CARAVAN SUIT
    --
    -- IMPORTANT:
    -- This is the current caravan suit.
    -- A Queen can change it.
    --------------------------------------------------------

    local suitOK =
        card.suit == caravan.suit


    --------------------------------------------------------
    -- EITHER CONDITION IS ENOUGH
    --------------------------------------------------------

    if directionOK or suitOK then

        return true

    end


    return false,
        "Card must continue the caravan's " ..
        "direction or match the caravan's suit."

end


------------------------------------------------------------
-- VALIDATE A NUMERICAL CARD
------------------------------------------------------------

function validateNumberCard(
    caravan,
    card
)

    --------------------------------------------------------
    -- FIRST CARD
    --------------------------------------------------------

    if #caravan.cards == 0 then

        return validateFirstCard(
            caravan,
            card
        )

    end


    --------------------------------------------------------
    -- SECOND CARD
    --------------------------------------------------------

    if #caravan.cards == 1 then

        return validateSecondCard(
            caravan,
            card
        )

    end


    --------------------------------------------------------
    -- THIRD+ CARD
    --------------------------------------------------------

    return validateSubsequentCard(
        caravan,
        card
    )

end

------------------------------------------------------------
-- GET CARAVAN CARD SNAP POSITION
------------------------------------------------------------

function getCaravanSnapPosition(
    caravanNumber,
    player,
    slot
)

    local counterGUID =
        COUNTERS[caravanNumber][player]

    local counter =
        getObjectFromGUID(counterGUID)


    if not counter then

        print(
            "ERROR: Counter not found: " ..
            counterGUID
        )

        return nil

    end


    local counterPosition =
        counter.getPosition()


    local offsetX
    local offsetZ
    local spacing


    --------------------------------------------------------
    -- PLAYER 1
    --------------------------------------------------------

    if player == 1 then

        offsetX =
            P1_SNAP_OFFSET_X

        offsetZ =
            P1_SNAP_OFFSET_Z

        spacing =
            P1_SNAP_SPACING


    --------------------------------------------------------
    -- PLAYER 2
    --------------------------------------------------------

    else

        offsetX =
            P2_SNAP_OFFSET_X

        offsetZ =
            P2_SNAP_OFFSET_Z

        spacing =
            P2_SNAP_SPACING

    end


    --------------------------------------------------------
    -- Calculate position
    --
    -- slot 1 = first position
    -- slot 2 = one spacing away
    -- etc.
    --
    -- We intentionally don't enforce SNAP_COUNT here.
    -- This allows physical overflow beyond slot 10.
    --------------------------------------------------------

    return {
        x =
            counterPosition.x +
            offsetX,

        y = CARAVAN_CARD_Y,

        z =
            counterPosition.z +
            offsetZ +
            (spacing * (slot - 1))
    }

end

------------------------------------------------------------
-- ADD CARD TO CARAVAN
------------------------------------------------------------

function addCardToCaravan(
    caravanNumber,
    player,
    card
)

    local caravan =
        GAME.caravans[caravanNumber][player]


    --------------------------------------------------------
    -- VALIDATE
    --------------------------------------------------------

    local valid, reason =
        validateNumberCard(
            caravan,
            card
        )


    if not valid then

        return false, reason

    end


    --------------------------------------------------------
    -- DETERMINE NEXT SLOT
    --------------------------------------------------------

    local slot =
        #caravan.cards + 1


    --------------------------------------------------------
    -- FIRST CARD ESTABLISHES SUIT
    --------------------------------------------------------

    if #caravan.cards == 0 then

        caravan.suit =
            card.suit

    end


    --------------------------------------------------------
    -- SECOND CARD ESTABLISHES DIRECTION
    --------------------------------------------------------

    if #caravan.cards == 1 then

        if card.rank >
           caravan.cards[1].rank then

            caravan.direction =
                "ascending"

        else

            caravan.direction =
                "descending"

        end

    end


    --------------------------------------------------------
    -- STORE SLOT AND GUID
    --------------------------------------------------------

    card.slot = slot
    card.guid = card.object.getGUID()


    --------------------------------------------------------
    -- ADD CARD
    --------------------------------------------------------

    table.insert(
        caravan.cards,
        card
    )


    --------------------------------------------------------
    -- MOVE CARD TO ITS CARAVAN SLOT
    --------------------------------------------------------

    local position =
        getCaravanSnapPosition(
            caravanNumber,
            player,
            slot
        )


    if position then

        card.object.setPositionSmooth(
            position
        )

    end


    --------------------------------------------------------
    -- RECALCULATE
    --------------------------------------------------------

    updateCaravanValue(
        caravanNumber,
        player
    )


    return true

end

function testMove(
    player,
    caravanNumber,
    cardOrGUID
)

    local cardObject


    --------------------------------------------------------
    -- ACCEPT EITHER AN OBJECT OR A GUID
    --------------------------------------------------------

    if type(cardOrGUID) == "string" then

        cardObject =
            getObjectFromGUID(cardOrGUID)

    else

        cardObject =
            cardOrGUID

    end


    --------------------------------------------------------
    -- CHECK OBJECT
    --------------------------------------------------------

    if not cardObject then

        print(
            "Could not find card."
        )

        return

    end


    --------------------------------------------------------
    -- PARSE CARD
    --------------------------------------------------------

    local card, errorMessage =
        parseCard(cardObject)


    if not card then

        print(
            "CARD ERROR: " ..
            errorMessage
        )

        return

    end


    --------------------------------------------------------
    -- ATTEMPT MOVE
    --------------------------------------------------------

    print(
        "Attempting to play " ..
        card.name ..
        " to Caravan " ..
        caravanNumber
    )


    local valid, reason =
        addCardToCaravan(
            caravanNumber,
            player,
            card
        )


    --------------------------------------------------------
    -- RESULT
    --------------------------------------------------------

    if valid then

        print(
            "LEGAL MOVE: " ..
            card.name
        )

        printCaravan(
            caravanNumber,
            player
        )

    else

        print(
            "ILLEGAL MOVE: " ..
            reason
        )

    end

end

function identifyCaravanZone(zone)

    for caravan = 1, 3 do

        for player = 1, 2 do

            if ZONES[caravan][player] == zone.getGUID() then

                return caravan, player

            end

        end

    end

    return nil, nil

end

function onObjectEnterScriptingZone(zone, object)

    local caravanNumber, player =
        identifyCaravanZone(zone)

    if not caravanNumber then
        return
    end

    if not isCaravanCard(object) then
        return
    end

    local guid = object.getGUID()

    -- Don't process the same card twice.
    if GAME.processing[guid] then
        return
    end

    if GAME.registeredCards[guid] then
        return
    end

    GAME.processing[guid] = true

    -- Give the player a moment to finish placing the card.
    Wait.time(
        function()

            processCaravanCard(
                zone,
                object,
                caravanNumber,
                player
            )

        end,
        0.05
    )

end


function isCaravanCard(object)

    if not object then
        return false
    end

    -- Ignore decks.
    if object.tag == "Deck" then
        return false
    end

    -- Ignore bags, counters, etc.
    if object.tag ~= "Card" then
        return false
    end

    -- Try to parse the card.
    local card = parseCard(object)

    if not card then
        return false
    end

    return true

end

function processCaravanCard(
    zone,
    object,
    caravanNumber,
    player
)

    local guid = object.getGUID()

    if GAME.gameOver then

    rejectCard(
        object,
        caravanNumber,
        player,
        "The game is already over."
    )

    GAME.processing[object.getGUID()] = nil

    --------------------------------------------------------
    -- Make sure the object still exists.
    --------------------------------------------------------

    if not object then

        GAME.processing[guid] = nil

        return

    end


    --------------------------------------------------------
    -- Make sure the card is STILL in this zone.
    --------------------------------------------------------

    local stillInZone = false

    for _, zoneObject in
        ipairs(zone.getObjects()) do

        if zoneObject.getGUID() == guid then

            stillInZone = true

            break

        end

    end


    if not stillInZone then

        GAME.processing[guid] = nil

        return

    end


    --------------------------------------------------------
    -- Parse card.
    --------------------------------------------------------

    local card, errorMessage =
        parseCard(object)


    if not card then

        GAME.processing[guid] = nil

        return

    end


    --------------------------------------------------------
    -- Currently only numerical cards & Queens.
    --------------------------------------------------------
    -- KING: Attach to a number card to double its value
    --------------------------------------------------------

    if card.type == "king" then

        -- Find which slot this secondary position corresponds to
        local slot = getSlotFromPosition(caravanNumber, player, object.getPosition())
        local target = findNumberCardBySlot(caravanNumber, player, slot)

        if not target then
            rejectCard(
                object,
                caravanNumber,
                player,
                "King must be placed on a number card (Ace-10), not on Queens or empty space."
            )
            GAME.processing[guid] = nil
            return
        end

        -- Kings cannot be attached to Queens (only to number cards)
        if target.type ~= "number" then
            rejectCard(
                object,
                caravanNumber,
                player,
                "King must be placed on a number card (Ace-10), not on Queens."
            )
            GAME.processing[guid] = nil
            return
        end

        -- Add guid to king card
        card.guid = guid

        -- Attach the King to the target card
        local success, reason = attachKingToCard(target, card)
        if not success then
            rejectCard(
                object,
                caravanNumber,
                player,
                reason
            )
            GAME.processing[guid] = nil
            return
        end

        -- Register the King
        GAME.registeredCards[guid] = {
            caravan = caravanNumber,
            player = player,
            attachedTo = target.guid
        }

        -- Position the King at the secondary snap point above the target
        card.object.setPositionSmooth({
            x = object.getPosition().x,
            y = SECONDARY_SNAP_Y,
            z = object.getPosition().z
        })

        -- Update the caravan value
        updateCaravanValue(caravanNumber, player)

        print(
            "[CARAVAN] KING ATTACHED: " ..
            card.name ..
            " -> " ..
            target.name ..
            " (slot " ..
            tostring(target.slot) ..
            ")"
        )
        printCaravan(caravanNumber, player)

        GAME.processing[guid] = nil
        return

    end

    if card.type == "queen" then

    local valid, reason =
        addQueenToCaravan(
            caravanNumber,
            player,
            card
        )

    if valid then

        GAME.registeredCards[guid] = {
            caravan = caravanNumber,
            player = player
        }

        GAME.processing[guid] = nil

        print(
            "[CARAVAN] LEGAL QUEEN: " ..
            card.name ..
            " → Player " ..
            player ..
            ", Caravan " ..
            caravanNumber
        )

        printCaravan(
            caravanNumber,
            player
        )

    else

        rejectCard(
            object,
            caravanNumber,
            player,
            reason
        )

        GAME.processing[guid] = nil

    end

    return

end



    end


    --------------------------------------------------------
    -- JACK: Remove target number card and any attached Kings
    --------------------------------------------------------

    if card.type == "jack" then

        -- Find which slot this secondary position corresponds to
        local slot = getSlotFromPosition(caravanNumber, player, object.getPosition())
        local target = findNumberCardBySlot(caravanNumber, player, slot)

        if not target then
            rejectCard(
                object,
                caravanNumber,
                player,
                "Jack must be placed on a number card (Ace-10)."
            )
            GAME.processing[guid] = nil
            return
        end

        -- Jacks can only be placed on number cards
        if target.type ~= "number" then
            rejectCard(
                object,
                caravanNumber,
                player,
                "Jack must be placed on a number card (Ace-10)."
            )
            GAME.processing[guid] = nil
            return
        end

        -- Add guid to jack card
        card.guid = guid

        -- Remove the target card and any attached Kings
        local success = removeCardAndAttachments(caravanNumber, player, target)
        if not success then
            rejectCard(
                object,
                caravanNumber,
                player,
                "Failed to remove target card."
            )
            GAME.processing[guid] = nil
            return
        end

        -- Send the removed target card to discard pile
        local targetObj = getObjectFromGUID(target.guid)
        if targetObj then
            sendCardToDiscard(targetObj, player)
        end

        -- Send any attached Kings to discard pile
        if GAME.attachedKings and GAME.attachedKings[target.guid] then
            for _, kingGuid in ipairs(GAME.attachedKings[target.guid]) do
                local kingObj = getObjectFromGUID(kingGuid)
                if kingObj then
                    sendCardToDiscard(kingObj, player)
                end
            end
        end

        -- Send the Jack itself to discard pile after activation
        sendCardToDiscard(card.object, player)

        -- Register the Jack (it was used to remove, so we track it)
        GAME.registeredCards[guid] = {
            caravan = caravanNumber,
            player = player,
            action = "removed",
            removedCard = target.guid
        }

        -- Update the caravan value
        updateCaravanValue(caravanNumber, player)

        print(
            "[CARAVAN] JACK REMOVED: " ..
            card.name ..
            " removed " ..
            target.name ..
            " (slot " ..
            tostring(target.slot) ..
            ") from Player " ..
            player ..
            " caravan " ..
            caravanNumber
        )
        printCaravan(caravanNumber, player)

        GAME.processing[guid] = nil
        return

    end


    --------------------------------------------------------
    -- Only number cards can be added normally
    --------------------------------------------------------

if card.type ~= "number" then

    rejectCard(
        object,
        caravanNumber,
        player,
        "This card type is not implemented yet."
    )

    GAME.processing[guid] = nil

    return

end


    --------------------------------------------------------
    -- Attempt to add card.
    --------------------------------------------------------

    local valid, reason =
        addCardToCaravan(
            caravanNumber,
            player,
            card
        )


    if valid then

        ----------------------------------------------------
        -- LEGAL
        ----------------------------------------------------

        GAME.registeredCards[guid] = {
            caravan = caravanNumber,
            player = player
        }

        GAME.processing[guid] = nil

        print(
            "[CARAVAN] LEGAL: " ..
            card.name ..
            " → Player " ..
            player ..
            ", Caravan " ..
            caravanNumber
        )

        printCaravan(
            caravanNumber,
            player
        )


    else

        ----------------------------------------------------
        -- ILLEGAL
        ----------------------------------------------------

        rejectCard(
            object,
            caravanNumber,
            player,
            reason
        )

        GAME.processing[guid] = nil

    end

end

function rejectCard(
    object,
    caravanNumber,
    player,
    reason
)

    print(
        "[CARAVAN] ILLEGAL MOVE: " ..
        reason
    )

    -- Return the card to the player's hand
    returnToHand(object, player)

end

function getCaravanStatus(value)

    if value == 0 then
        return "empty"
    end

    if value >= 21 and value <= 26 then
        return "sold"
    end

    if value > 26 then
        return "overloaded"
    end

    return "building"

end

function updateCaravanAppearance(
    caravanNumber,
    player
)

    local caravan =
        GAME.caravans[caravanNumber][player]

    local counter =
        getCounter(
            caravanNumber,
            player
        )

    if not counter then
        return
    end


    if caravan.status == "overloaded" then

        -- Red = overloaded
        counter.setColorTint({
            r = 1,
            g = 0.2,
            b = 0.2
        })


    elseif caravan.status == "sold" then

        -- Green = currently in winning range
        counter.setColorTint({
            r = 0.3,
            g = 1,
            b = 0.3
        })


    else

        -- Normal
        counter.setColorTint({
            r = 1,
            g = 1,
            b = 1
        })

    end

end

function getLaneWinner(caravanNumber)

    local p1 =
        GAME.caravans[caravanNumber][1]

    local p2 =
        GAME.caravans[caravanNumber][2]


    --------------------------------------------------------
    -- Neither caravan is in selling range.
    --------------------------------------------------------

    if p1.status ~= "sold"
       and p2.status ~= "sold" then

        return nil

    end


    --------------------------------------------------------
    -- P1 is sold, P2 isn't.
    --------------------------------------------------------

    if p1.status == "sold"
       and p2.status ~= "sold" then

        return 1

    end


    --------------------------------------------------------
    -- P2 is sold, P1 isn't.
    --------------------------------------------------------

    if p2.status == "sold"
       and p1.status ~= "sold" then

        return 2

    end


    --------------------------------------------------------
    -- Both are in range.
    -- Higher value wins.
    --------------------------------------------------------

    if p1.value > p2.value then

        return 1

    elseif p2.value > p1.value then

        return 2

    end


    --------------------------------------------------------
    -- Same value = contested.
    --------------------------------------------------------

    return nil

end

function checkWinner()

    if GAME.gameOver then
        return
    end


    local wins = {
        [1] = 0,
        [2] = 0
    }


    --------------------------------------------------------
    -- Check all three lanes.
    --------------------------------------------------------

    for caravan = 1, 3 do

        local winner =
            getLaneWinner(caravan)

        if winner then

            wins[winner] =
                wins[winner] + 1

        end

    end


    --------------------------------------------------------
    -- Nobody has two lanes yet.
    --------------------------------------------------------

    if wins[1] < 2
       and wins[2] < 2 then

        return

    end


    --------------------------------------------------------
    -- Game over.
    --------------------------------------------------------

    GAME.gameOver = true


    local winner

    if wins[1] >= 2 then

        winner = 1

    else

        winner = 2

    end


    announceWinner(
        winner,
        wins[1],
        wins[2]
    )

end

function announceWinner(
    winner,
    p1Wins,
    p2Wins
)

    local color

    if winner == 1 then
        color = "Blue"
    else
        color = "Red"
    end


    broadcastToAll(
        "CARAVAN GAME OVER!\n" ..
        color ..
        " wins " ..
        math.max(p1Wins, p2Wins) ..
        " caravans!",
        {
            1,
            0.8,
            0.2
        }
    )


    print(
        "================================"
    )

    print(
        "CARAVAN GAME OVER"
    )

    print(
        "Player 1 wins: " ..
        p1Wins
    )

    print(
        "Player 2 wins: " ..
        p2Wins
    )

    print(
        "Winner: Player " ..
        winner
    )

    print(
        "================================"
    )

end

function resetCaravanState()

    print("[CARAVAN] Resetting caravan state...")

    for caravanNumber = 1, 3 do

        for player = 1, 2 do

            local caravan =
                GAME.caravans[caravanNumber][player]

            caravan.cards = {}
            caravan.suit = nil
            caravan.direction = nil
            caravan.value = 0
            caravan.status = "empty"

            updateCounter(
                caravanNumber,
                player
            )

            updateCaravanAppearance(
                caravanNumber,
                player
            )

        end

    end

    -- Clear all attachment tracking
    GAME.attachedKings = {}
    GAME.registeredCards = {}

    -- Clear discard piles
    DISCARD_PILES = {
        [1] = {},
        [2] = {}
    }

    print("[CARAVAN] Caravan state reset.")

end

    if message == "!resetcaravan" then

        broadcastToAll(

        )

    end

end

------------------------------------------------------------
-- ADD QUEEN TO CARAVAN
------------------------------------------------------------

function addQueenToCaravan(
    caravanNumber,
    player,
    card
)

    local caravan =
        GAME.caravans[caravanNumber][player]


    if not caravan then

        return false,
            "Caravan does not exist."

    end


    --------------------------------------------------------
    -- A Queen cannot be the first card.
    --------------------------------------------------------

    if #caravan.cards == 0 then

        return false,
            "A Queen cannot be played on an empty caravan."

    end


    --------------------------------------------------------
    -- DETERMINE NEXT SLOT
    --------------------------------------------------------

    local slot =
        #caravan.cards + 1


    --------------------------------------------------------
    -- QUEEN GETS A SLOT AND GUID
    --
    -- But it does NOT target another card.
    --------------------------------------------------------

    card.slot = slot
    card.guid = card.object.getGUID()


    --------------------------------------------------------
    -- ADD QUEEN TO CARAVAN
    --------------------------------------------------------

    table.insert(
        caravan.cards,
        card
    )


    --------------------------------------------------------
    -- MOVE QUEEN TO ITS CARAVAN SLOT
    --------------------------------------------------------

    local position =
        getCaravanSnapPosition(
            caravanNumber,
            player,
            slot
        )


    if position then

        card.object.setPositionSmooth(
            position
        )

    end


    --------------------------------------------------------
    -- QUEEN CHANGES CARAVAN SUIT
    --------------------------------------------------------

    caravan.suit =
        card.suit


    --------------------------------------------------------
    -- QUEEN REVERSES CARAVAN DIRECTION
    --------------------------------------------------------

    if caravan.direction == "ascending" then

        caravan.direction =
            "descending"

    elseif caravan.direction == "descending" then

        caravan.direction =
            "ascending"

    end


    --------------------------------------------------------
    -- RECALCULATE CARAVAN
    --------------------------------------------------------

    updateCaravanValue(
        caravanNumber,
        player
    )


    return true

end

------------------------------------------------------------
-- GENERATE CARAVAN SNAP POINTS
------------------------------------------------------------

function generateCaravanSnapPoints()

    local snapPoints = {}


    --------------------------------------------------------
    -- THREE CARAVANS
    --------------------------------------------------------

    for caravanNumber = 1, 3 do


        ----------------------------------------------------
        -- TWO PLAYERS
        ----------------------------------------------------

        for player = 1, 2 do

            local counterGUID =
                COUNTERS[caravanNumber][player]

            local counter =
                getObjectFromGUID(counterGUID)


            if counter == nil then

                print(
                    "ERROR: Counter not found: " ..
                    counterGUID
                )

            else

                local counterPosition =
                    counter.getPosition()


                ------------------------------------------------
                -- Select player's layout
                ------------------------------------------------

                local offsetX
                local offsetZ
                local spacing


                if player == 1 then

                    offsetX =
                        P1_SNAP_OFFSET_X

                    offsetZ =
                        P1_SNAP_OFFSET_Z

                    spacing =
                        P1_SNAP_SPACING

                else

                    offsetX =
                        P2_SNAP_OFFSET_X

                    offsetZ =
                        P2_SNAP_OFFSET_Z

                    spacing =
                        P2_SNAP_SPACING

                end


                ------------------------------------------------
                -- Generate 10 positions
                ------------------------------------------------

                for slot = 0, SNAP_COUNT - 1 do

                    local x =
                        counterPosition.x +
                        offsetX

                    local z =
                        counterPosition.z +
                        offsetZ +
                        (spacing * slot)


                    table.insert(
                        snapPoints,
                        {
                            position = {
                                x,
                                0,
                                z
                            }
                        }
                    )

                end

            end

        end

    end


    --------------------------------------------------------

    --------------------------------------------------------

    --------------------------------------------------------
    -- Generate SECONDARY snap points for face cards (King, Jack, Joker)
    -- These are positioned above each primary snap point
    -- When a face card is placed on a secondary point, it targets the card below
    --------------------------------------------------------

    for caravanNumber = 1, 3 do
        for player = 1, 2 do
            for slot = 0, SNAP_COUNT - 1 do
                local baseIndex = ((caravanNumber - 1) * 2 + (player - 1)) * SNAP_COUNT + slot + 1
                
                -- Create secondary snap point above each primary point
                local secondaryIndex = #snapPoints + 1
                table.insert(
                    snapPoints,
                    {
                        position = {
                            snapPoints[baseIndex].position[1],
                            SECONDARY_SNAP_Y,
                            snapPoints[baseIndex].position[3]
                        },
                        -- Store metadata for identifying this as a secondary point
                        caravan = caravanNumber,
                        player = player,
                        slot = slot,
                        isSecondary = true
                    }
                )
            end
        end
    end

    Global.setSnapPoints(snapPoints)


    print(
        "[CARAVAN] Generated " ..
        #snapPoints ..
        " snap points."
    )

end

------------------------------------------------------------
-- GET NEXT CARAVAN SLOT
------------------------------------------------------------

function getNextCaravanSlot(caravan)

    if not caravan or not caravan.cards then
        return 1
    end

    return #caravan.cards + 1

end
